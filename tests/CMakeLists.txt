cmake_minimum_required(VERSION 3.20)
project(PrismMaterialXTests LANGUAGES CXX)
find_package(MaterialX REQUIRED)
enable_testing()
add_executable(materialx_loader_sss materialx_loader_sss.cpp ../src/materialx/loader.cpp)
target_compile_features(materialx_loader_sss PRIVATE cxx_std_20)
target_link_libraries(materialx_loader_sss PRIVATE MaterialXCore MaterialXFormat)
add_test(NAME materialx_loader_sss COMMAND materialx_loader_sss)
add_executable(materialx_loader_clamp materialx_loader_clamp.cpp ../src/materialx/loader.cpp)
target_compile_features(materialx_loader_clamp PRIVATE cxx_std_20)
target_link_libraries(materialx_loader_clamp PRIVATE MaterialXCore MaterialXFormat)
add_test(NAME materialx_loader_clamp COMMAND materialx_loader_clamp)
add_executable(materialx_loader_library materialx_loader_library.cpp ../src/materialx/loader.cpp)
target_compile_features(materialx_loader_library PRIVATE cxx_std_20)
target_link_libraries(materialx_loader_library PRIVATE MaterialXCore MaterialXFormat)
add_test(NAME materialx_loader_library COMMAND materialx_loader_library)
add_executable(materialx_loader_types materialx_loader_types.cpp ../src/materialx/loader.cpp)
target_compile_features(materialx_loader_types PRIVATE cxx_std_20)
target_link_libraries(materialx_loader_types PRIVATE MaterialXCore MaterialXFormat)
add_test(NAME materialx_loader_types COMMAND materialx_loader_types)
add_executable(materialx_loader_transmission materialx_loader_transmission.cpp ../src/materialx/loader.cpp)
target_compile_features(materialx_loader_transmission PRIVATE cxx_std_20)
target_link_libraries(materialx_loader_transmission PRIVATE MaterialXCore MaterialXFormat)
add_test(NAME materialx_loader_transmission COMMAND materialx_loader_transmission)
add_executable(materialx_loader_fresnel materialx_loader_fresnel.cpp ../src/materialx/loader.cpp)
target_compile_features(materialx_loader_fresnel PRIVATE cxx_std_20)
target_link_libraries(materialx_loader_fresnel PRIVATE MaterialXCore MaterialXFormat)
add_test(NAME materialx_loader_fresnel COMMAND materialx_loader_fresnel)
# Checks the repairs against MaterialX's own library graphs, from the
# MaterialX install (skipped without them).
add_executable(materialx_loader_dielectric materialx_loader_dielectric.cpp ../src/materialx/loader.cpp)
target_compile_features(materialx_loader_dielectric PRIVATE cxx_std_20)
target_link_libraries(materialx_loader_dielectric PRIVATE MaterialXCore MaterialXFormat)
get_filename_component(_prism_materialx_root "${MaterialX_DIR}/../../.." ABSOLUTE)
target_compile_definitions(materialx_loader_dielectric PRIVATE PRISM_TEST_MATERIALX_ROOT="${_prism_materialx_root}")
add_test(NAME materialx_loader_dielectric COMMAND materialx_loader_dielectric)
set_tests_properties(materialx_loader_dielectric PROPERTIES SKIP_RETURN_CODE 77)

# CPU-only; see the labels in the top-level CMakeLists.txt.
get_property(_prism_materialx_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${_prism_materialx_tests} PROPERTIES LABELS cpu)
